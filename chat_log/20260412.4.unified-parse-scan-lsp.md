# Session: Unified Parse + Scan + LSP Architecture (Zoom 3)

## Goal
Create a unified analysis pipeline that combines parse, lower, and extract phases with shared error recovery, enabling rich LSP features (goto-definition, diagnostics, hover) while keeping the LSP layer thin.

## Current State
- Partial parsing implemented (`parse_program_partial`)
- In-memory extraction working (`extract_in_memory`, `extract_from_content`)
- LSP diagnostics partially working (generated but not always visible)
- File pattern hover with clickable links implemented
- Unsaved file support for extraction implemented

## Problem/Context
LSP currently does too much work inline:
1. Calls `parse_program()` separately
2. Calls `scan_file()` separately  
3. Does its own diagnostic generation
4. Has no symbol table for goto-definition

This violates "parser as shell" philosophy and duplicates logic.

/// Complete analysis result with all phases
pub struct PartialAnalysis {
    pub source: String,
    pub stmts: Vec<Statement>,
    pub rules: Vec<Rule>,
    pub extractions: Vec<RuleExtraction>,
    pub errors: Vec<AnalysisError>,
    pub diagnostics: Vec<AnalysisDiagnostic>,
    pub symbol_table: SymbolTable,
}

pub struct AnalysisError {
    pub phase: Phase,
    pub location: SourceRange,
    pub message: String,
    pub severity: Severity,
}

#[derive(Clone, Copy, Debug)]
pub enum Phase { Parse, Lower, Extract }

#[derive(Clone, Copy, Debug)]
pub enum Severity { Error, Warning, Info }

/// Source location for diagnostics and goto-definition
#[derive(Clone, Debug)]
pub struct SourceRange {
    pub start: usize,
    pub end: usize,
    pub line: usize,
    pub column: usize,
}

/// Symbol table for LSP goto-definition
pub struct SymbolTable {
    pub rules: HashMap<String, SymbolLocation>,
    pub captures: HashMap<(String, String), SymbolLocation>,
    pub cross_refs: HashMap<String, Vec<SourceRange>>,
}

#[derive(Clone, Debug)]
pub struct SymbolLocation {
    pub range: SourceRange,
    pub kind: SymbolKind,
}

#[derive(Clone, Copy, Debug)]
pub enum SymbolKind {
    RuleDefinition,
    CaptureDefinition,
    RuleReference,
    CaptureReference,
    CrossRuleReference,
}

pub struct AnalyzeOptions {
    pub run_extraction: bool,
    pub workspace_root: PathBuf,

/// Main entry point - unified analysis
pub fn analyze_partial(source: &str, options: AnalyzeOptions) -> PartialAnalysis {
    // Phase 1: Parse with error recovery
    let (stmts, parse_errors, stmt_spans) = parse_phase(source);
    
    // Phase 2: Lower with per-statement error recovery
    let (rules, lower_errors, symbol_table) = lower_phase(&stmts, &stmt_spans, source);
    
    // Phase 3: Extract with per-rule error recovery (optional)
    let (extractions, extract_diagnostics) = if options.run_extraction {
        extract_phase(&rules, &options)
    } else {
        (vec![], vec![])
    };
    
    // Combine all errors
    let mut errors = vec![];
    errors.extend(parse_errors);
    errors.extend(lower_errors);
    
    PartialAnalysis {
        source: source.to_string(),
        stmts,
        rules,
        extractions,
        errors,
        diagnostics: extract_diagnostics,
        symbol_table,
    }
}

fn parse_phase(source: &str) -> (Vec<Statement>, Vec<AnalysisError>, Vec<SourceRange>) {
    use crate::_1_parse::parse_statement;
    
    let mut stmts = vec![];
    let mut errors = vec![];
    let mut spans = vec![];
    let mut remaining = source;
    let mut offset = 0;
    
    while !remaining.is_empty() {
        remaining = skip_ws_and_comments(remaining);
        if remaining.is_empty() { break; }
        
        match parse_statement(remaining) {
            Ok((stmt, rest)) => {
                let stmt_len = remaining.len() - rest.len();
                spans.push(SourceRange {
                    start: offset,
                    end: offset + stmt_len,
                    line: line_number(source, offset),
                    column: column_number(source, offset),
                });
                stmts.push(stmt);
                offset += stmt_len;
                remaining = rest;
            }
            Err(e) => {
                errors.push(AnalysisError {
                    phase: Phase::Parse,
                    location: SourceRange {
                        start: offset,
                        end: offset + remaining.len().min(50),
                        line: line_number(source, offset),
                        column: column_number(source, offset),
                    },
                    message: e.to_string(),
                    severity: Severity::Error,
                });
                // Recovery: skip to next statement boundary
                let recovery_pos = find_recovery_point(remaining);
                offset += recovery_pos;
                remaining = &remaining[recovery_pos..];
            }
        }
    }
    
    (stmts, errors, spans)
}

fn lower_phase(
    stmts: &[Statement],
    spans: &[SourceRange],
    source: &str,
) -> (Vec<Rule>, Vec<AnalysisError>, SymbolTable) {
    use crate::_3_lower::lower_rule;
    
    let mut rules = vec![];
    let mut errors = vec![];
    let mut symbols = SymbolTable {
        rules: HashMap::new(),
        captures: HashMap::new(),
        cross_refs: HashMap::new(),
    };
    
    for (idx, stmt) in stmts.iter().enumerate() {
        let span = spans.get(idx).cloned().unwrap_or_else(|| SourceRange {
            start: 0, end: 0, line: 0, column: 0,
        });
        
        match stmt {
            Statement::Rule(decl) => {
                match lower_rule(decl) {
                    Ok(rule) => {
                        // Record rule symbol
                        symbols.rules.insert(
                            rule.name.clone(),
                            SymbolLocation {
                                range: span.clone(),
                                kind: SymbolKind::RuleDefinition,
                            }
                        );
                        
                        // Record capture symbols from create_matches
                        for match_def in &rule.create_matches {
                            symbols.captures.insert(
                                (rule.name.clone(), match_def.capture.clone()),
                                SymbolLocation {
                                    range: span.clone(), // Could be more precise
                                    kind: SymbolKind::CaptureDefinition,
                                }
                            );
                        }
                        
                        rules.push(rule);
                    }
                    Err(e) => {
                        errors.push(AnalysisError {
                            phase: Phase::Lower,
                            location: span,
                            message: format!("Failed to lower rule '{}': {}", decl.name, e),
                            severity: Severity::Error,
                        });
                    }
                }
            }
            Statement::Check(decl) => {
                // Check statements don't produce rules
                // Could record check symbols here if needed
            }
        }
    }
    
    (rules, errors, symbols)
}

fn extract_phase(
    rules: &[Rule],
    options: &AnalyzeOptions,
) -> (Vec<RuleExtraction>, Vec<AnalysisDiagnostic>) {
    let mut extractions = vec![];
    let mut diagnostics = vec![];
    
    for rule in rules {
        // Find matching files
        let files = crate::_4_extract::find_matching_files(rule, &options.workspace_root);
        
        if files.is_empty() {
            diagnostics.push(AnalysisDiagnostic {
                rule: rule.name.clone(),
                message: format!("Rule '{}' matched 0 files - check file pattern", rule.name),
                severity: Severity::Error,
            });
            continue;
        }
        
        // Extract from each file
        let mut total_captures = 0;
        let mut file_results = vec![];
        
        for file_path in files.iter().take(50) {
            let uri = file_path_to_uri(file_path);
            
            // Get content (unsaved or from disk)
            let content = options.documents.get(&uri)
                .cloned()
                .or_else(|| std::fs::read_to_string(file_path).ok());
            
            let Some(content) = content else {
                continue;
            };
            
            // Extract captures for each variable
            for match_def in &rule.create_matches {
                let var_name = &match_def.capture;
                let values = extract_from_content(rule, var_name, file_path, content.as_bytes());
                total_captures += values.len();
                file_results.push((file_path.clone(), var_name.clone(), values));
            }
        }
        
        if total_captures == 0 {
            // Check if it's a JSON path issue
            let path_issue = check_json_path_mismatch(rule, &files[0], options);
            
            if path_issue {
                diagnostics.push(AnalysisDiagnostic {
                    rule: rule.name.clone(),
                    message: format!(
                        "Rule '{}' found {} files but JSON/YAML path doesn't match file structure",
                        rule.name, files.len()
                    ),
                    severity: Severity::Error,
                });
            } else {
                diagnostics.push(AnalysisDiagnostic {
                    rule: rule.name.clone(),
                    message: format!(
                        "Rule '{}' found {} files but extracted 0 captures - check capture variables",
                        rule.name, files.len()
                    ),
                    severity: Severity::Warning,
                });
            }
        }
        
        extractions.push(RuleExtraction {
            rule_name: rule.name.clone(),
            files_scanned: files.len(),
            results: file_results,
        });
    }
    
    (extractions, diagnostics)
}

// LSP Support: Find symbol at position for goto-definition
pub fn find_symbol_at_position(analysis: &PartialAnalysis, offset: usize) -> Option<SymbolAtCursor> {
    // Check if offset is within a rule name
    for (name, loc) in &analysis.symbol_table.rules {
        if loc.range.start <= offset && offset <= loc.range.start + name.len() {
            return Some(SymbolAtCursor {
                name: name.clone(),
                kind: CursorSymbolKind::RuleDefinition,
            });
        }
    }
    
    // Check for capture variable references ($VAR)
    let source = &analysis.source;
    if offset < source.len() && source[offset..].starts_with('$') {
        // Extract var name
        let after_dollar = &source[offset+1..];
        let var_end = after_dollar.find(|c: char| !c.is_ascii_alphanumeric() && c != '_')
            .unwrap_or(after_dollar.len());
        let var_name = &after_dollar[..var_end];
        
        // Find which rule we're in
        if let Some(rule_name) = find_rule_at_offset(analysis, offset) {
            return Some(SymbolAtCursor {
                name: format!("{}.{}", rule_name, var_name),
                kind: CursorSymbolKind::CaptureReference,
            });
        }
    }
    
    // Check for cross-rule references (rule_name(...))
    // Implementation would scan for Ident followed by LParen
    
    None
}

pub struct SymbolAtCursor {
    pub name: String,
    pub kind: CursorSymbolKind,
}

pub enum CursorSymbolKind {
    RuleDefinition,
    RuleReference,
    CaptureDefinition,
    CaptureReference,
}

// LSP Support: Get definition location for symbol
pub fn get_definition_location(
    analysis: &PartialAnalysis,
    symbol: &SymbolAtCursor,
) -> Option<SymbolLocation> {
    match symbol.kind {
        CursorSymbolKind::RuleDefinition | CursorSymbolKind::RuleReference => {
            analysis.symbol_table.rules.get(&symbol.name).cloned()
        }
        CursorSymbolKind::CaptureReference => {
            // Parse "rule.var" format
            let parts: Vec<_> = symbol.name.split('.').collect();
            if parts.len() == 2 {
                analysis.symbol_table.captures.get(&(parts[0].to_string(), parts[1].to_string())).cloned()
            } else {
                None
            }
        }
        _ => None,
    }
}

// LSP Support: Convert to LSP diagnostics
pub fn to_lsp_diagnostics(analysis: &PartialAnalysis) -> Vec<lsp_types::Diagnostic> {
    let mut diags = vec![];
    
    for err in &analysis.errors {
        diags.push(lsp_types::Diagnostic {
            range: source_range_to_lsp(&analysis.source, &err.location),
            severity: Some(match err.severity {
                Severity::Error => lsp_types::DiagnosticSeverity::ERROR,
                Severity::Warning => lsp_types::DiagnosticSeverity::WARNING,
                Severity::Info => lsp_types::DiagnosticSeverity::INFORMATION,
            }),
            code: None,
            code_description: None,
            source: Some(format!("sprf-{:?}", err.phase)),
            message: err.message.clone(),
            related_information: None,
            tags: None,
            data: None,
        });
    }
    
    for diag in &analysis.diagnostics {
        diags.push(lsp_types::Diagnostic {
            range: lsp_types::Range::default(), // Would find rule position
            severity: Some(match diag.severity {
                Severity::Error => lsp_types::DiagnosticSeverity::ERROR,
                Severity::Warning => lsp_types::DiagnosticSeverity::WARNING,
                Severity::Info => lsp_types::DiagnosticSeverity::INFORMATION,
            }),
            code: None,
            code_description: None,
            source: Some("sprf-extract".to_string()),
            message: format!("[{}] {}", diag.rule, diag.message),
            related_information: None,
            tags: None,
            data: None,
        });
    }
    
    diags
}

fn find_recovery_point(input: &str) -> usize {
    // Find next rule( or check( keyword
    for keyword in ["rule(", "check("] {
        if let Some(pos) = input.find(keyword) {
            if pos > 0 { return pos; }
        }
    }
    // Or find semicolon
    input.find(';').map(|p| p + 1).unwrap_or(input.len())
}

fn file_path_to_uri(path: &Path) -> String {
    url::Url::from_file_path(path)
        .map(|u| u.to_string())
        .unwrap_or_else(|_| path.to_string_lossy().to_string())
}

fn source_range_to_lsp(source: &str, range: &SourceRange) -> lsp_types::Range {
    // Convert byte offsets to LSP Position (line, column)
    let start_line = line_number(source, range.start) - 1;
    let start_col = column_number(source, range.start);
    let end_line = line_number(source, range.end) - 1;
    let end_col = column_number(source, range.end);
    
    lsp_types::Range {
        start: lsp_types::Position::new(start_line as u32, start_col as u32),
        end: lsp_types::Position::new(end_line as u32, end_col as u32),
    }
}

use std::collections::HashMap;
use std::sync::Mutex;
use std::time::Instant;
use sprefa_sprf::analyze::PartialAnalysis;

pub struct LspState {
    pub cache: Mutex<AnalysisCache>,
    pub workspace: Mutex<Option<crate::workspace::Workspace>>,
    pub documents: Mutex<HashMap<String, String>>, // uri -> unsaved content
}

pub struct AnalysisCache {
    analyses: HashMap<String, CachedAnalysis>,
}

pub struct CachedAnalysis {
    pub analysis: PartialAnalysis,
    pub version: i32,
    pub timestamp: Instant,
}

impl LspState {
    pub fn new() -> Self {
        Self {
            cache: Mutex::new(AnalysisCache { analyses: HashMap::new() }),
            workspace: Mutex::new(None),
            documents: Mutex::new(HashMap::new()),
        }
    }
    
    pub fn get_analysis(&self, uri: &str) -> Option<PartialAnalysis> {
        let cache = self.cache.lock().unwrap();
        cache.analyses.get(uri).map(|c| c.analysis.clone())
    }
    
    pub fn cache_analysis(&self, uri: &str, analysis: PartialAnalysis, version: i32) {
        let mut cache = self.cache.lock().unwrap();
        cache.analyses.insert(uri.to_string(), CachedAnalysis {
            analysis,
            version,
            timestamp: Instant::now(),
        });
    }
}

// In on_change:
async fn on_change(&self, uri: &Url, text: &str, version: i32) {
    let workspace_root = self.workspace.lock().unwrap()
        .as_ref().map(|w| w.root.clone())
        .unwrap_or_else(|| std::env::current_dir().unwrap());
    
    let documents = self.documents.lock().unwrap().clone();
    
    let analysis = sprefa_sprf::analyze::analyze_partial(
        text,
        sprefa_sprf::analyze::AnalyzeOptions {
            run_extraction: true,
            workspace_root,
            documents,
        }
    );
    
    // Cache for other requests
    self.state.cache_analysis(&uri.to_string(), analysis.clone(), version);
    
    // Publish diagnostics
    let diags = sprefa_sprf::analyze::to_lsp_diagnostics(&analysis);
    self.client.publish_diagnostics(uri.clone(), diags, None).await;
}

// In goto_definition:
async fn goto_definition(&self, params: GotoDefinitionParams) -> Result<Option<GotoDefinitionResponse>> {
    let uri = params.text_document_position_params.text_document.uri;
    let pos = params.text_document_position_params.position;
    
    let analysis = self.state.get_analysis(&uri.to_string())
        .ok_or_else(|| tower_lsp::jsonrpc::Error::invalid_params("No analysis cached"))?;
    
    let offset = position_to_offset(&analysis.source, pos);
    let symbol = sprefa_sprf::analyze::find_symbol_at_position(&analysis, offset)
        .ok_or_else(|| tower_lsp::jsonrpc::Error::invalid_params("No symbol at position"))?;
    
    let def = sprefa_sprf::analyze::get_definition_location(&analysis, &symbol)
        .ok_or_else(|| tower_lsp::jsonrpc::Error::invalid_params("No definition found"))?;
    
    Ok(Some(GotoDefinitionResponse::Scalar(Location {
        uri: uri::Url::parse(&uri.to_string()).unwrap(),
        range: source_range_to_lsp_range(&def.range),
    })))
}
